LOCAL SIGHT()
BEGIN  
  HAngle:=1;

  // Check that our variables are initialized
  IF TYPE(sights)=0 THEN RESET_SIGHTS(); END;

  IF NOT INPUT_HS_AND_BODY() THEN RETURN "CANCELED"; END;
  
  LOCAL c;
  
  CASE 
    IF sights(current,i_sightType)=1 THEN c:=SUN_SIGHT(); END;
    IF sights(current,i_sightType)=2 THEN c:=MOON_SIGHT(); END;
    IF sights(current,i_sightType)=3 THEN c:=STAR_SIGHT(); END;
    IF sights(current,i_sightType)=4 THEN c:=PLANET_SIGHT(); END;
    IF sights(current,i_sightType)=7 THEN
      c:=PRE_CALCULATED_SIGHT();
      // As the name implies, this is a pre-calculated
      // sight so no need to continue with the HC
      // and LOP calculations.
      IF NOT c THEN RETURN "CANCELED"; ELSE RETURN c; END;
    END;
  END;
  
  IF NOT c THEN RETURN "CANCELED"; END;
  
  IF NOT HC_CALC(current) THEN RETURN "CANCELED"; END;

  IF NOT VIEW_LOP_PARAMETERS(current) THEN RETURN "CANCELED"; END;
  
  LOP_EQUATION(current);
    
END;

LOCAL SUN_SIGHT()
BEGIN
  HAngle:=1;

  IF progressFlag THEN
    PRINT();
    PRINT("Calculating heliocentric Earth ephemeris...");
  END;
  
  LOCAL sunData:=SunEphemeris(sights(current,i_date),sights(current,i_time),progressFlag);
  
  sights(current,i_declination):=sunData(2);
  sights(current,i_GHA):=sunData(1);

  HO_CALC_SUN(current,sunData(4)/60);
  
  RETURN 1;
END;

LOCAL MOON_SIGHT()
BEGIN
  HAngle:=1;
    
  // Add a semi-diameter correction
  // for lower limb and subtract it for upper
  sights(current,i_limb):=-1*((sights(current,i_limbSign)-1)*2-1);
  
  IF progressFlag THEN
    PRINT();
    PRINT("Calculating Moon ephemeris...");    
  END;
  
  LOCAL r:=MoonEphemeris(sights(current,i_date),sights(current,i_time));

  sights(current,i_GHA):=r(1);
  sights(current,i_declination):=r(2);

  // True altitude
  sights(current,i_HO):=sights(current,i_HA)-REFRACTION_MINUTES(current)/60;
  
  // Corrected altitude including semi-diameter
  // correction and parallax correction
  sights(current,i_HO):=sights(current,i_HO)
    +sights(current,i_limb)*r(4)/3600
    +r(3)/3600*COS(sights(current,i_HO));
  
  RETURN 1;
END;

 
LOCAL PLANET_SIGHT()
BEGIN
  HAngle:=1;
 
  // Calculate altitude correction.
  // NOTE: the planet ephemeris function
  // returns the position of
  // the centroid of the illuminated disc for Venus
  // and Mars, so the Additional Correction in the
  // Almanac for Mars and Venus MUST NOT be applied.
  
  sights(current,i_altitudeCorr_min):=-1*REFRACTION_MINUTES(current);
  
  sights(current,i_HO):=sights(current,i_HA)+sights(current,i_altitudeCorr_min)/60;
  
  IF progressFlag THEN
    PRINT();
    PRINT("Calculating planet ephemeris...");
  END;  
  
  LOCAL r:=PlanetEphemeris(
             sights(current,i_subBodyIndex),
             sights(current,i_date),
             sights(current,i_time),
             1,
             progressFlag);
  
  sights(current,i_declination):=r(2);
  sights(current,i_SHA):=360-r(5);
  sights(current,i_GHA):=r(1);

  RETURN 1;
END;

LOCAL STAR_SIGHT()
BEGIN  
  HAngle:=1;
  
  sights(current,i_altitudeCorr_min):=-1*REFRACTION_MINUTES(current);
  sights(current,i_HO):=sights(current,i_HA)+sights(current,i_altitudeCorr_min)/60;
 
  IF progressFlag THEN
    PRINT();
    PRINT("Calculating star ephemeris...");
  END;

  LOCAL r:=StarEphemeris(
             sights(current,i_subBodyIndex),
             sights(current,i_date),
             sights(current,i_time));
       
  sights(current,i_declination):=r(2);
  sights(current,i_SHA):=360-r(3);
  sights(current,i_GHA):=r(1);

  RETURN 1;
END;

LOCAL PRE_CALCULATED_SIGHT()
BEGIN

  LOCAL c:=INPUT({sights(current,i_a_nm),sights(current,i_Zn)},
        "LOP "+STRING(current)+" Pre-calculated values",
        {"A:","ZN:"},
        {"Distance from assumed position (nm)",
         "Azimuth to celestial body (°)"});
  
  sights(current,i_HS):=1e-12;
  
  IF NOT c THEN RETURN c; END;

  RETURN LOP_EQUATION(current);
  
END;

LOCAL CHECK_LOP_EXISTS()
BEGIN
  // Check that our variables are initialized
  IF TYPE(sights)=0 THEN RESET_SIGHTS(); END;

  // Check that we have at least one LOP defined
  LOCAL c;
  FOR c FROM 1 TO 10 DO
    IF sights(c,i_a_nm)<>0 THEN
      c:=15;
      BREAK;
    END;
  END;
  
  RETURN c==15;
END;


LOCAL PLOT_LOPs()
BEGIN
  HAngle:=1;

  IF NOT CHECK_LOP_EXISTS() THEN MSGBOX("No LOPs available"); RETURN -1; END;

  Function.Xmin:=-180;
  Function.Xmax:=180;
  Function.Ymin:=-90;
  Function.Ymax:=90;
  LOCAL i:=1;
  
  STARTAPP("Function");
  
  FOR i FROM 0 TO 9 DO  
    IF sights(IFTE(i=0,10,i),i_a_nm)<>0 THEN Function.CHECK(i); ELSE Function.UNCHECK(i); END;
  END;
  
  STARTVIEW(1,1);
  
  RETURN 0;
END;

LOCAL INTERSECTION(f1,f2)
BEGIN
  LOCAL xx, yy;
  
  xx:=EXPR("CAS(\"solve(Function.F"+f1+"(x)=Function.F"+f2+"(x),x)(1)\")");
  yy:=EXPR("CAS(\"Function.F"+f1+"("+xx+")\")");

  RETURN {xx,yy};
END;

LOCAL LEMOINE(p1,p2,p3)
BEGIN
  // pn are triangle vertices {xn,yn}
  
  LOCAL x1 := p1(1), y1 := p1(1);
  LOCAL x2 := p2(1), y2 := p2(2);
  LOCAL x3 := p3(1), y3 := p3(2);
  
  LOCAL a2 := (x3-x2)^2 + (y3-y2)^2;
  LOCAL b2 := (x3-x1)^2 + (y3-y1)^2;
  LOCAL c2 := (x2-x1)^2 + (y2-y1)^2;
  
  LOCAL S := a2 + b2 + c2;
  
  LOCAL xl := (a2*x1 + b2*x2 + c2*x3) / S;
  LOCAL yl := (a2*y1 + b2*y2 + c2*y3) / S;
  
  RETURN {xl, yl};
END;

LOCAL COMPUTE_FIX()
BEGIN
  LOCAL i,fix;
  LOCAL lops={};
  LOCAL msg:="";

  HAngle:=1;
  
  STARTAPP("Function");
  STARTVIEW(-7,1);

  // Check that variables are initialized
  IF SIZE(sights)==0 THEN RETURN "Variables not initialized."; END;
  
  // Determine which LOP registers
  // contain a non-translated LOP
  FOR i FROM 1 TO 10 DO
    IF sights(i,i_a_nm)<>0 THEN
    IF i==10 THEN
      lops:=CONCAT(lops,0);
    ELSE   
      lops:=CONCAT(lops,i);
    END;
  END;
  END;

  CASE
    IF SIZE(lops)<2 THEN MSGBOX("At least 2 LOPs are required"); RETURN; END;
  
  IF SIZE(lops)==2 THEN
    // 2 LOPs, we return the intersection of the 2
    fix:=INTERSECTION(lops(1),lops(2));
    msg:="Intersection of 2 LOPs";
  END;
  
  IF SIZE(lops)==3 THEN
    // 3 LOPs: compute symmedian
    // center of triangle (Lemoine point)
    fix:=LEMOINE(
      INTERSECTION(lops(1),lops(2)),
    INTERSECTION(lops(1),lops(3)),
    INTERSECTION(lops(2),lops(3)));
    
    msg:="Symmedian Center of 3 LOPs";
  END;
  
  IF SIZE(lops)>3 THEN
    // With more than 3 LOPs, we compute the
    // coordinate with the lowest mean square
    // error from all LOPs
    
      
    // Start by creating our mean square
    // function which is the sum of the
    // square distance of a point (x,y) from
    // all LOPs
    CAS("PURGE(x,y)");
    LOCAL eqn:="Fsq(x,y):=0";
      FOR i FROM 1 TO SIZE(lops) DO
        LOCAL m:=EXPR("CAS(\"diff(Function.F"+lops(i)+"(x),x)\")");
        LOCAL b:=EXPR("CAS(\"Function.F"+lops(i)+"(0)\")");
        EXPR("CAS(\"Fsq"+i+"(x,y):=("+m+"*x-y+"+b+")^2/("+m+"^2+1)\")");
        eqn:=eqn+"+Fsq"+i+"(x,y)";
      END;
      EXPR("CAS(\""+eqn+"\")");
    
    // Find the minimum mean square by
    // differentiating the mean square function
    // along x and y and finding the coordinate(s)
      // where the derivatives are both zero
      fix:=CAS("solve({diff(Fsq(x,y),x)=0,diff(Fsq(x,y),y)=0},{x,y})");

      // In case there are multiple minima,
      // take the one with the lowest mean square value
      LOCAL minimumValue:=0,minimumIndex;
      FOR i FROM 1 TO SIZE(fix) DO
        LOCAL value:=EXPR("Fsq("+fix(i,1)+","+fix(i,2)+")");
        IF value<minimumValue OR i=1 THEN
          minimumValue:=value;
          minimumIndex:=i;
        END;
      END;
    
      // Cleanup
      CAS("PURGE(Fsq)");
      FOR i FROM 1 TO SIZE(lops) DO
        EXPR("CAS(\"PURGE(Fsq"+i+")\")");
      END;

      fix:={fix(minimumIndex,1),fix(minimumIndex,2)};
    msg:="Minimizing Mean Squares of "+SIZE(lops)+" LOPs";

  END;  
  END;
  
  fixLon:=fix(1);
  fixLat:=fix(2);
  
  PRINT();
  
  PRINT("Fix from "+msg+CHAR(10)+CHAR(10)
    +"   Latitude: "+→DMD(fixLat)+CHAR(10)
  +"   Longitude: "+→DMD(fixLon)+CHAR(10));
  
  i:=GET_SOFTKEY({"","","","","Check","Exit"});
  
  IF i==0 OR i==6 THEN 
    PRINT(); 
  PRINT("Press ESC to exit.");
  RETURN "Done."; 
  END;
  
  i:=INPUT({actualLat,actualLon},
    "Actual Position",
  {"Latitude","Longitude"},
  {"Latitude of actual position (°)",
     "Longitude of actual position (°)"});  
    
  IF i==0 THEN RETURN "Canceled."; END;

  PRINT();
  PRINT("COMPUTED FIX");
  PRINT("  Lat: "+→DMD(fixLat));
  PRINT("  Lon: "+→DMD(fixLon));
  PRINT(" ");
  PRINT("ACTUAL POSITION");
  PRINT("  Lat: "+→DMD(actualLat));
  PRINT("  Lon: "+→DMD(actualLon));
  PRINT(" ");
  PRINT("Distance between coordinates: "+STRING(ROUND(NM_DIST(fixLat,fixLon,actualLat,actualLon),1))+" nm");

  PRINT(" ");
  PRINT("Press any key to continue.");
  WAIT(-1);
END;

BEGIN
  LOCAL nl:=CHAR(10);

         "GHA : "+→DMD(sights(c,i_GHA))+nl+

         "LHA : "+→DMD(sights(c,i_LHA))+nl+

  IF NOT MSGBOX("LOP: " + c + nl
         + "a (nm): "+IFTE(ABS(sights(c,i_a_nm))<0.05,"0.0",STRING(HMS→(ABS(sights(c,i_a_nm))),2,1)
           +IFTE(sights(c,i_a_nm)≤0," AWAY"," TOWARD")) + nl
         + "Zn (°): "+STRING(sights(c,i_Zn),2,1) + nl
         + "AP lat (°): "+→DMD(sights(c,i_DRlat)) + nl
         + "AP lon (°): "+→DMD(sights(c,i_DRlon)),1) THEN RETURN 0; END;
  
  RETURN 1;
  
END;

LOCAL SHIFT_LOP()
BEGIN
  LOCAL f,index,fn;
  LOCAL d_nm,brg_deg;

  IF NOT CHECK_LOP_EXISTS() THEN MSGBOX("No LOPs available"); RETURN -1; END;

  HAngle:=1;
  STARTAPP("Function");
  STARTVIEW(-7,1);

  LOCAL c:=SELECT_LOP(current,"Select LOP to shift");  
  IF NOT c THEN RETURN c; END;
 
  LOCAL rv:=INPUT({sights(c,i_translationBearing),
         sights(c,i_translationDistance_nm)},
    "LOP "+STRING(c)+" Shift",
    {"Bearing:","Distance:"},
    {"Bearing direction for shift (°)",
     "Distance of shift (nm)"
     });
  IF NOT rv THEN RETURN rv; END;

  fn:=c;

  // If register 10 is chosen, it
  // must be retrieved from Function F0
  IF fn==10 THEN fn:=0; END;

  //@TODO: TEST ACCURACY!
  LOCAL newCoords:=atdz(sights(c,i_DRlat),sights(c,i_DRlon),sights(c,i_translationDistance_nm),sights(c,i_translationBearing));
  LOCAL dy:=newCoords(1)-sights(c,i_DRlat);
  LOCAL dx:=newCoords(2)-sights(c,i_DRlon);
  
  // Retrieve LOP formula and use x
  // instead of the reserved X (which always
  // has an assigned value) and add
  // offsets using CAS    
  
  // Make sure x is not set  
  CAS("PURGE(x)");

  //@TODO: Redo without EXPR -- see 
  //AVERAGE_READINGS function for example
  f:="CAS(\"Function.F"+STRING(c)+"(x-dx)+dy\");";
  f:=STRING(EXPR(f));

  // Replace x with X using string
  // manipulation to ensure X is not
  // evaluated to its assigned value
  index:=INSTRING(f,"x");
  f:=LEFT(f,index-1)+"'X'"
     + MID(f,index+1);

  LOCAL d:=SELECT_LOP(c,"Select destination LOP register");
  IF NOT d THEN RETURN d; END;
  IF d=10 THEN d:=0; END;

  // Clear a_nm value for source LOP so
  // it is no longer included in COMPUTE_FIX
  // calculations or PLOT_FIX results
  // Set destination bodies(d) to 
  // source LOP ID for reference
  sights(c,i_a_nm):=0;

  // Set a_nm to a non-zero value in the destination
  // LOP register so the shifted line will be displayed
  // in PLOT_LOPs
  sights(d,i_a_nm):=1e-12;
  sights(d,i_bodyIndex):=c;  

  // Store LOP equation in destination
  // formula register
  f:="Function.F"+STRING(d)+":="+f;

  EXPR(f);      
  
  RETURN "Shifted LOP #"+STRING(c)+" by "
         +→DMD(dy)+" lat. and "+→DMD(dx)+" lon."
         +" into LOP #"+STRING(d);
END;

LOCAL WRITELINE(lin,data,haFlag,riseSetFlag)
BEGIN
    LOCAL i:=1;
    LOCAL body:=data(1);
    LOCAL azimuth:=data(2);
    LOCAL altitude:=data(3);
    LOCAL gha:=data(4);
    LOCAL dec:=data(5);
    LOCAL sd:=data(6);
    LOCAL rise:=data(8);
    LOCAL set:=data(9);
 

    Spreadsheet.Cell(lin,i,1):=body;
    Spreadsheet.Cell(lin,i+1,1):=→DMD(azimuth);
    Spreadsheet.Cell(lin,i+2,1):=→DMD(altitude);
    IF haFlag THEN
      Spreadsheet.Cell(lin,i+3,1):=→DMD(altitude+REFRACTION_MINUTES(1010,20,altitude)/60);
    ELSE
      i:=i-1;
    END;
    Spreadsheet.Cell(lin,i+4,1):=→DMD(gha);
    Spreadsheet.Cell(lin,i+5,1):=→DMD(dec);
    IF sd<>0 THEN Spreadsheet.Cell(lin,i+6,1):=ROUND(sd,1); END; 
    
    IF riseSetFlag THEN
      IF (rise>0) AND (set>0) THEN
        Spreadsheet.Cell(lin,i+7,1):=HMS(rise);
        Spreadsheet.Cell(lin,i+8,1):=HMS(set);
      ELSE
        Spreadsheet.Cell(lin,i+7,1):="-";
        Spreadsheet.Cell(lin,i+8,1):="-";
      END;      
    END;
    
    RETURN 1;
END;

CALCULATE_BODIES(date,time,lat,lon,flags)
BEGIN
  LOCAL result:={};
  LOCAL r,i;

  // Parse flags
  LOCAL sunFlag := flags(1);
  LOCAL moonFlag := flags(2);
  LOCAL planetFlag := flags(3);
  LOCAL starFlag := flags(4);
  LOCAL haFlag := flags(5);
  LOCAL includeHeading := flags(6);
  LOCAL showProgress := flags(7);
  LOCAL sortFlag := flags(8);
  LOCAL riseSetFlag := flags(9);
  
  IF sunFlag THEN
    IF showProgress THEN PRINT("Computing sun"); END;
    r:=SunEphemeris(date,time,showProgress);    
    r:=CONCAT(r,RiseSetTime(lat,lon,r(2),0.833));
    result:=CONCAT(result,{CONCAT({"Sun"},r)});    
  END;

  IF moonFlag THEN
    IF showProgress THEN PRINT("Computing Moon"); END;
    r:=MoonEphemeris(date,time);
    r:=CONCAT(r,RiseSetTime(lat,lon,r(2),-0.125));
    result:=CONCAT(result,{CONCAT({"Moon"},r)});
  END;

  IF planetFlag THEN
    FOR i FROM 1 TO 4 DO        
      r:=SUB(PlanetEphemeris(i,date,time,1,showProgress),1,4);
      r:=CONCAT(r,RiseSetTime(lat,lon,r(2),0.5667));
      result:=CONCAT(result,{CONCAT({planets_(i)},r)});      
    END;
  END;

  IF starFlag THEN
    IF showProgress THEN PRINT("Computing stars"); END;
    LOCAL d:=StarEphemerisAll(date,time);

    FOR i FROM 1 TO SIZE(d) DO      
      r:=CONCAT(SUB(d(i),1,2),{0,0});
      r:=CONCAT(r,RiseSetTime(lat,lon,r(2),0.5667));
      result:=CONCAT(result,{CONCAT({navstar_(i,1)},r)});      
    END;
  END;

  // Calculate azimuth and elevation of all bodies
  FOR i FROM 1 TO SIZE(result) DO
    LOCAL c:=HC_AZ(result(i,2),result(i,3),lat,lon);
    LOCAL altitude:=c(1);
    LOCAL azimuth:=c(2);
    result(i):={result(i,1),azimuth,altitude,result(i,2),result(i,3),result(i,4),result(i,5),result(i,6),result(i,7)};
  END;
  
  RETURN result;
END;


OUTPUT_EPHEMERIDES(date,time,lat,lon,minAltitude,minAz,maxAz,
  flags,startingLine)
BEGIN
  LOCAL i,c,lin:=startingLine,r;
  LOCAL result:={};

  // Parse flags
  LOCAL sunFlag := flags(1);
  LOCAL moonFlag := flags(2);
  LOCAL planetFlag := flags(3);
  LOCAL starFlag := flags(4);
  LOCAL haFlag := flags(5);
  LOCAL includeHeading := flags(6);
  LOCAL showProgress := flags(7);
  LOCAL sortFlag := flags(8);
  LOCAL riseSetFlag := flags(9);

  IF includeHeading THEN
    c:=2;
    Spreadsheet.Cell(lin,c,1):="Azimuth";  
    Spreadsheet.Cell(lin,c+1,1):="Alt. (Hc)";
    IF haFlag THEN
      Spreadsheet.Cell(lin,c+2,1):="App Alt (HA)";
    ELSE
      c:=c-1;
    END;
    Spreadsheet.Cell(lin,c+3,1):="GHA";
    Spreadsheet.Cell(lin,c+4,1):="Dec";
    Spreadsheet.Cell(lin,c+5,1):="Semi D";
    IF riseSetFlag THEN
      Spreadsheet.Cell(lin,c+6,1):="Rise";
      Spreadsheet.Cell(lin,c+7,1):="Set";
    END;
    
    FOR i FROM 1 TO IFTE(riseSetFlag,8,6)+haFlag DO 
      Spreadsheet.Cell(lin,i,7):=IFTE(Theme(1)==1,16753920,#500000);
    END;  
    lin:=lin+1;
  END;

  LOCAL result:=CALCULATE_BODIES(date,time,lat,lon,flags);

  //Restrict azimuth and elevation of all bodies
  LOCAL r:={};
  
  FOR i FROM 1 TO SIZE(result) DO
    LOCAL altitude:=result(i,3);
    LOCAL azimuth:=result(i,2);

    IF altitude>=minAltitude THEN
      IF ((minAz<maxAz) AND (azimuth>=minAz) AND (azimuth<=maxAz))
         OR ((minAz>maxAz) AND ((azimuth>=minAz) OR (azimuth<=maxAz))) THEN
        r:=CONCAT(r,{result(i)});
      END;
    END;
  END;

  IF sortFlag THEN
    //Sort by azimuth
    result:=SORT(r,2);

    // Adjust list if azimuth range crosses north
    // to show all results in clockwise order
    IF minAz>maxAz THEN
      i:=1;
      WHILE i<SIZE(result) DO
        IF result(i,2)>=minAz THEN
          IF i>1 THEN
            result:=CONCAT(SUB(result,i,SIZE(result)),
              SUB(result,1,i-1));
          END;
          BREAK;
        END;
        i:=i+1;
      END;
    END;
  ELSE
    result:=r;
  END;

  FOR i FROM 1 TO SIZE(result) DO  
    lin:=lin+WRITELINE(lin,result(i),haFlag,riseSetFlag);
  END;

  RETURN lin;
END;

LOCAL prevViewAz:=9999,prevViewAlt:=9999,COSviewAz,SINviewAz,COSviewAlt,SINviewAlt;

LOCAL AZALT2SCREENSPACE(az, alt, viewAz, viewAlt_, fov)
BEGIN
  LOCAL screenWidth := 320; 
  LOCAL screenHeight := 240;

  LOCAL x_, y_, z_, p_;
  LOCAL xp, yp, zp, p_;
  
  LOCAL viewAlt:=90-viewAlt_;
  
  // Convert relative azimuth and altitude to
  // 3D points on a unit sphere
  x_:=SIN(az)*COS(alt);
  y_:=COS(az)*COS(alt);
  z_:=SIN(alt);

  IF viewAz<>prevViewAz OR viewAlt<>prevViewAlt THEN
    COSviewAz:=COS(viewAz);
    SINviewAz:=SIN(viewAz);
    COSviewAlt:=COS(viewAlt);
    SINviewAlt:=SIN(viewAlt);
    prevViewAz:=viewAz;
    prevViewAlt:=viewAlt;
  END;

  // Rotate to align with observer gaze
  xp:=x_*COSviewAz-y_*SINviewAz;
  p_:=x_*SINviewAz+y_*COSviewAz;
  yp:=-1*(p_*COSviewAlt-z_*SINviewAlt);
  zp:=p_*SINviewAlt+z_*COSviewAlt;
  
  // Apply the stereographic projection  
  IF zp <> -1 THEN
    // Calculate zoom factor to display desired FOV
    z_:=COT(fov/4);
    x_:=xp/(1+zp)*z_;
    y_:=yp/(1+zp)*z_;
    IF ABS(x_)<1.5 AND ABS(y_)<1.5 THEN
      RETURN {(x_+1)/2*screenWidth,(1-y_)/2*screenHeight};
    END;
  END;
  
  RETURN {-1,-1};
  
END;

// Project celestial bodies on a perspective view
// for an observer looking at a given azimuth and
// altitude.
LOCAL CELESTIALTOSCREEN(bodies, viewAz, viewAlt, fov)
BEGIN
  LOCAL result := {};
  LOCAL i, body, name, az, alt;
  LOCAL camX, camY, camZ;
  LOCAL screenX, screenY;
  LOCAL tanHalfFov;
    
  // Process each celestial body
  FOR i FROM 1 TO SIZE(bodies) DO
    body := bodies(i);
    name := body(1);
    az := body(2);    // Already in degrees
    alt := body(3);   // Already in degrees
    
    LOCAL r:=AZALT2SCREENSPACE(az,alt,viewAz,viewAlt,fov);

    IF NOT(r(1)=-1 AND r(2)=-1) THEN 
      // Add to result list
      result := CONCAT(result, {{name,r(1),r(2),body(2),body(3)}});
    END;
  END;
  
  RETURN result;
END;

LOCAL PLOTSKYGRID(viewAz,viewAlt,fov,azStep,altStep,density_,color)
BEGIN
  LOCAL i,j;
    
  LOCAL start:=FLOOR((viewAz-fov/2)/azStep)*azStep;
  LOCAL end:=CEILING((viewAz+fov/2)/azStep)*azStep;
  LOCAL mid:=FLOOR((end+start)/2/azStep)*azStep;

  //Adjust density based on FOV
  i:=1;
  IF fov<40 THEN 
    i:=1/2;
    IF fov<20 THEN
      i:=1/4;
      IF fov<10 THEN
        i:=1/8;
      END;
    END;
  END;
  LOCAL density:=density_*i;

  //Find out the first visible altitude line to
  //determine where azimuth labels will be
  //printed
  LOCAL altLabels:=-90,coords;
  REPEAT
    altLabels:=altLabels+altStep;
    coords:=AZALT2SCREENSPACE(start+azStep,altLabels,viewAz,viewAlt,fov);
  UNTIL coords(1)<>-1 OR altLabels>=90;

  LOCAL coords,screenX,screenY,prevX:=-1,prevY:=-1,labelPrinted;
 
  // Plot lines of constant azimuth
  FOR i FROM 0 TO 179 STEP azStep DO
    prevX:=-1; prevY:=-1;
    FOR j FROM 0 to 360+density STEP density DO
      LOCAL coords,screenX,screenY;
      coords:=AZALT2SCREENSPACE(i,j,viewAz,viewAlt,fov);
      screenX:=coords(1);
      screenY:=coords(2);

      IF NOT(screenX==-1 AND screenY==-1) THEN
        IF NOT(prevX==-1 AND prevY==-1) THEN
          LINE_P(prevX,prevY,screenX,screenY,color);
        END;
      END;
      prevX:=screenX;
      prevY:=screenY;
    END;    
    coords:=AZALT2SCREENSPACE(i,0,viewAz,viewAlt,fov);
    IF coords(1)<>-1 THEN
      TEXTOUT_P(i,coords(1)+2,coords(2)-10,1,color);
    END;
    IF i<180 THEN
      coords:=AZALT2SCREENSPACE(i+180,0,viewAz,viewAlt,fov);
      IF coords(1)<>-1 THEN
        TEXTOUT_P(i+180,coords(1)+2,coords(2)-10,1,color);
      END;
    END;
  END;
  
  // Plot lines of constant altitude
  FOR i FROM -90+altStep TO 90-altStep STEP altStep DO
    prevX:=-1; prevY:=-1;
    labelPrinted:=0;
    FOR j FROM 0 TO 360+density STEP density DO
      coords:=AZALT2SCREENSPACE(j,i,viewAz,viewAlt,fov);
      screenX:=coords(1);
      screenY:=coords(2);
    
      IF NOT(screenX==-1 AND screenY==-1) THEN
        IF NOT(prevX==-1 AND prevY==-1) THEN
          LINE_P(prevX,prevY,screenX,screenY,color);
          IF i==0 THEN
            LINE_P(prevX,prevY-1,screenX,screenY-1,color);
            LINE_P(prevX,prevY+1,screenX,screenY+1,color);
          END;
          IF (NOT labelPrinted) AND j>mid THEN
            coords:=AZALT2SCREENSPACE(mid,i,viewAz,viewAlt,fov);
            LOCAL labelX:=coords(1);
            coords:=AZALT2SCREENSPACE(mid+azStep,i,viewAz,viewAlt,fov);
            labelX:=(labelX+coords(1))/2-10;
            TEXTOUT_P(i,labelX,screenY-10,1,color);
            labelPrinted:=1;
          END;
        END;
      END;
      prevX:=screenX;
      prevY:=screenY;
    END;
  END;
END;

// Plots the output of the CELESTIALTOSCREEN function
// on the calculator screen
LOCAL PLOTBODIES(screenCoords)
BEGIN
  LOCAL i, body, name, screenX, screenY, color;
  LOCAL pixelX, pixelY;
  
  // Draw each celestial body
  FOR i FROM 1 TO SIZE(screenCoords) DO
    body := screenCoords(i);
    name := body(1);
    pixelX := body(2);
    pixelY := body(3);
    // Draw bodies in red if below horizon
    // blue if above in light theme, aqua
    // in dark theme
    IF body(5)<0 THEN
      color:=#FF0000;
    ELSE
      color:=IFTE(Theme(1)==1,#3037FF,#00FFFF);
    END;

    // Draw a 3-pixel wide dot
    PIXON_P(pixelX - 1, pixelY - 1,color);
    PIXON_P(pixelX, pixelY - 1,color);
    PIXON_P(pixelX + 1, pixelY - 1,color);
    PIXON_P(pixelX - 1, pixelY,color);
    PIXON_P(pixelX, pixelY,color);
    PIXON_P(pixelX + 1, pixelY,color);
    PIXON_P(pixelX - 1, pixelY + 1,color);
    PIXON_P(pixelX, pixelY + 1,color);
    PIXON_P(pixelX + 1, pixelY + 1,color);
    
    // Draw the label to the right in small font
    TEXTOUT_P(name, pixelX + 5, pixelY - 3, 1);
  END;
END;

// Plots the sky and processes user input to
// tilt and pan the view using keys or the mouse
LOCAL PLOTSKY(bodies,viewAz,viewAlt,viewFov)
BEGIN
  LOCAL az:=viewAz;
  LOCAL alt:=viewAlt;
  LOCAL fov:=viewFov;
  LOCAL i,k:=0;
  LOCAL validKeys:={2,7,8,12,4,45,50,46};
  LOCAL oldX:=-1,oldY:=-1,mx,my,dx,dy;
  LOCAL redrawRequired:=1;
  LOCAL viewStep:=5;

  REPEAT
    //Monitor key and mouse inputs in a loop
    //Exit loop if a key is pressed or the mouse
    //is dragged
    k:=-1;
    WHILE k==-1 DO
      WAIT(0.05);

      // Check whether a key is pressed
      FOR i FROM 1 TO SIZE(validKeys) DO
        IF ISKEYDOWN(validKeys(i)) THEN
          k:=validKeys(i);
          BREAK;
        END;
      END;

      //Draw stars and gridlines
      IF redrawRequired THEN
        RECT();
        TEXTOUT_P("Az:"+az+"° Alt:"+alt+"° FOV:"+fov+"°",2,230,1);
        PLOTBODIES(CELESTIALTOSCREEN(bodies,az,alt,fov));        
        // Only redraw gridlines if panning has stopped
        LOCAL m:=MOUSE();
        IF SIZE(m(1))==0 AND k==-1 THEN
          PLOTSKYGRID(az,alt,fov,15,10,15,IFTE(Theme(1)==1,#CCCCCC,#555555));
          TEXTOUT_P("Az:"+az+"° Alt:"+alt+"° FOV:"+fov+"°",2,230,1);
          // Redraw bodies in case they were covered by grid lines
          PLOTBODIES(CELESTIALTOSCREEN(bodies,az,alt,fov));        
          redrawRequired:=0;
        END;
        IF fov>80 THEN 
          viewStep:=15;
        ELSE 
          IF fov>40 THEN viewStep:=10; ELSE viewStep:=5; END;
        END;
      END;

      // Check if the mouse is being dragged
      LOCAL mouse:=MOUSE();
      mouse:=mouse(1);
      IF SIZE(mouse)>0 THEN
        mx:=mouse(1);
        my:=mouse(2);
        IF oldX<>-1 THEN
          dx:=mx-oldX;
          dy:=my-oldY;

          // If drag values are non-null, pan and tilt
          // the view by a factor corresponding to the
          // drag values
          alt:=alt+dy/5;
          az:=az-dx/5;
          redrawRequired:=1;
          k:=0;
        ELSE
          dx:=0;
          dy:=0;
        END;
        oldX:=mx;
        oldY:=my;
      ELSE
        oldX:=-1;
        oldY:=-1;
        dx:=0;
        dy:=0;
      END;
    END;

    // If we're here, a key was pressed or the
    // mouse was dragged

    // Respond to arrow and + or - key presses
    CASE
      // Arrows cause viewer azimuth and altitude
      // to change
      IF k=2 THEN alt:=alt+viewStep; redrawRequired:=1; END;
      IF k=12 THEN alt:=alt-viewStep; redrawRequired:=1; END;
      IF k=8 THEN az:=az+viewStep; redrawRequired:=1; END;
      IF k=7 THEN az:=az-viewStep; redrawRequired:=1; END;

      // + and - keys cause FOV to change
      IF k=50 THEN fov:=fov-5; redrawRequired:=1; END;
      IF k=45 THEN fov:=fov+5; redrawRequired:=1; END;
    END;
    
    // Check that values remain within bounds
    IF az > 360 THEN az := az - 360; ELSE
    IF az < 0 THEN az := az + 360; END; END;
    IF alt > 90 THEN alt := 90; ELSE
    IF alt < -90 THEN alt := -90; END; END;
    IF fov > 135 THEN fov:=135; ELSE
    IF fov < 5 THEN fov:=5; END; END;
    
  // Exit if On or Esc is pressed
  UNTIL k==4 OR k==46;
END;

LOP_EQUATION(index)
BEGIN
  LOCAL c:=atdz(sights(index,i_DRlat),sights(index,i_DRlon),
                sights(index,i_a_nm),sights(index,i_Zn));

  LOCAL f:=−1/TAN(90-sights(index,i_Zn))
            *COS(sights(index,i_DRlat))
            *('X'-c(2))+c(1);

  LOCAL i:=index;
  IF i=10 THEN i:=0; END;
  EXPR("Function.F"+i+":=f");
END;

AVERAGE_READINGS(deltaUTCTime)
BEGIN  
  LOCAL i:=1,more:=1,c,t,r;
  LOCAL hs:={},hourOffset:=0,scalingFactor:=10000;
  
  WHILE more DO
    RECT();
    TEXTOUT_P("OBSERVATION #"+i,0,20,0);
    TEXTOUT_P("Press any key to record the time",0,155,1);
    TEXTOUT_P("when the observation is made and",0,170,1);
    TEXTOUT_P("enter the sextant reading.",0,185,1);

    // We use TICKS() instead of reading the Time
    // variable because the Time variable sometimes
    // skips seconds
    LOCAL startTicks:=TICKS();
    LOCAL startTime:=GET_UTC_TIME();
  
  IF startTime<0 THEN
    startTime:=startTime+24;
  ELSE
    IF startTime>24 THEN
      startTime:=startTime-24;
    END;
  END;
  
    REPEAT
      TEXTOUT_P(HMS(startTime+(TICKS()-startTicks)/3600000),120,80,7,#FFFFFF,320,0);
      WAIT(0.5);  
    UNTIL GETKEY()<>-1;

    RECT();

    t:=GET_UTC_TIME();
    t:=t+deltaUTCTime;
    
    r:=0;
    c:=INPUT({r,t},"Observation #"+i,
      {"HS:","Time:"},
      {"Enter sextant reading (°), or 0 to end",
       "UTC Time (HH°MM'SS\")"});

    IF ((c=0) OR (r=0)) AND (i=1) THEN RETURN {}; END;
  
    IF r>0 THEN
      more:=1;
    // Remove hour part of time and expand
    // time readings to increase the
    // accuracy of statistical computations
    IF hourOffset=0 THEN hourOffset:=IP(t); END;
    t:=scalingFactor*(t-hourOffset);  
      hs:=CONCAT(hs,{{t,r}});
      i:=i+1;
    ELSE
      i:=i-1;
      more:=0
    END;  
  END;
  
  // If only one observation was entered,
  // we return that value
  
  IF i=1 THEN     
    RETURN {hs(1,1)/scalingFactor+hourOffset,hs(1,2)};
  END;
  
  // Otherwise we perform a regression
  // and interpolate a value on a regression
  // curve, and display the results
  
  STARTAPP("Statistics_2Var");
  Statistics_2Var.C1:={};
  Statistics_2Var.C2:={};
  
  FOR t FROM 1 TO i DO
    Statistics_2Var.C1:=CONCAT(Statistics_2Var.C1,{hs(t,1)});
    Statistics_2Var.C2:=CONCAT(Statistics_2Var.C2,{hs(t,2)});
  END;
  
  // Select linear regression if 3 or fewer
  // data points, cubic otherwise
  IF SIZE(Statistics_2Var.C1)<4 THEN r:=1; ELSE r:=9; END;
  
  Statistics_2Var.Fit:=1;
  Statistics_2Var.S1:={"C1","C2","",r,"",
    IFTE(Theme(1)==1,#FF,6545151),4,IFTE(Theme(1)==1,0,1675320)};
  Statistics_2Var.CHECK(1);
  
  // Compute graph zoom factors
  LOCAL factor:=(MAX(Statistics_2Var.C1)-MIN(Statistics_2Var.C1))*1.1;  
  Statistics_2Var.Xmax:=MAX(Statistics_2Var.C1)+factor;
  Statistics_2Var.Xmin:=MIN(Statistics_2Var.C1)-factor;
  
  factor:=(MAX(Statistics_2Var.C2)-MIN(Statistics_2Var.C2))*1.1;
  Statistics_2Var.Ymax:=MAX(Statistics_2Var.C2)+factor;
  Statistics_2Var.Ymin:=MIN(Statistics_2Var.C2)-factor;
    
  LOCAL fitOptions:={
    "Linear",
    IFTE(i>2,"Quad",""), 
    IFTE(i>3,"Cubic",""),
    IFTE(i>3,"Trig",""),
    "Cancel",
    "Ok"};
  LOCAL fitTypes:={1,8,9,11};

  LOCAL avgTime,avgHs;
      
  REPEAT
  // Determine interpolated sight. If
  // the regression function has an extremum
  // within our observed time range, use that
  // extremum as our interpolated point.
  // Otherwise, use the midpoint of the time
  // range.
  
  // Extremums only exist for quadratic or
  // cubic regressions
  LOCAL extremumFlag:=0;
    IF Statistics_2Var.S1(4)=8 OR Statistics_2Var.S1(4)=9 THEN
    // Find the extremums of the regression function    
    r:=CAS("solve(eval(diff(subst(Statistics_2Var.S1(5),'X'=x),x))=0,x)");         
    // Check that they fall within our time range
    IF SIZE(r)>0 THEN
      FOR i FROM 1 TO SIZE(r) DO
        IF (r(i)>Statistics_2Var.C1(1)) AND (r(i)<Statistics_2Var.C1(0)) THEN
          extremumFlag:=1;
          avgTime:=r(i);
        END;
      END;
    ELSE
      avgTime:=Statistics_2Var.C1(1)+(Statistics_2Var.C1(0)-Statistics_2Var.C1(1))/2;
      END;
  ELSE
    avgTime:=Statistics_2Var.C1(1)+(Statistics_2Var.C1(0)-Statistics_2Var.C1(1))/2;
  END;
  avgHs:=PredY(avgTime);

  Statistics_2Var.C3:={avgTime,avgTime};
  Statistics_2Var.C4:={avgHs,avgHs};
  Statistics_2Var.S2:={"C3","C4","",13,"",
    IFTE(Theme(1)==1,#FF,6545151),2,#FF0000};
  Statistics_2Var.CHECK(2);
    
  // Show graph and results
  STARTVIEW(1,1);
  TEXTOUT_P("Review results and click Ok to continue",G0,1,10,1,IFTE(Theme(1)==1,0,#FFFFFF));
    TEXTOUT_P("Interpolated values: "+HMS(avgTime/scalingFactor+hourOffset)+" - "+→DMD(avgHs)
    +IFTE(extremumFlag," (extremum)"," (midpoint)"),G0,1,25,1,IFTE(Theme(1)==1,0,#FFFFFF));
    
  // Monitor soft keys and include
  // indicator next to currently
  // selected key
  LOCAL flags:={0,0,0,0,0,0};
  flags(POS(fitTypes,Statistics_2Var.S1(4))):=1;
    c:=GET_SOFTKEY(fitOptions,flags);
  IF c=5 OR c=0 THEN 
    STARTVIEW(-7,1); 
    RETURN {};
  END;
  IF c>0 AND c<5 THEN
    IF fitOptions(c)<>"" THEN
      Statistics_2Var.S1:={"C1","C2","",fitTypes(c),"",
        IFTE(Theme(1)==1,#FF,6545151),4,IFTE(Theme(1)==1,0,1675320)};
        Statistics_2Var.CHECK(1);
      STARTVIEW(1,1);        
    END;
  END;
  UNTIL c=6;

  // Go back to program view (where
  // the user likely was when the program
  // was started)
  STARTVIEW(-7,1);

  RETURN {→HMS(avgTime/scalingFactor)+hourOffset,→HMS(avgHs)}; 
END;

INPUT_HS_AND_BODY()
BEGIN
  LOCAL deltaUTCTime:=0;
  
  LOCAL c:=SELECT_LOP(current,"Select LOP register",1);
  IF NOT c THEN RETURN c; END;

  current:=c;

  // OBSERVATION PARAMETERS
  LOCAL tempTime:=sights(current,i_time);
  LOCAL tempDate:=sights(current,i_date);

  IF tempDate==0 THEN 
    tempDate:=Date; 
  END;
    
  IF tempTime=0 THEN 
    tempTime:=GET_UTC_TIME();
    IF tempTime>24 THEN 
      tempTime:=tempTime-24;
      deltaUTCTime:=-24;
      tempDate:=DATEADD(tempDate,1);
    END;
    IF tempTime<0 THEN 
      tempTime:=tempTime+24;
      deltaUTCTime:=24;
      tempDate:=DATEADD(tempDate,-1);
    END;
  END;

  IF sights(current,i_DRlat)=0 THEN sights(current,i_DRlat):=DRlat_default; END;
  IF sights(current,i_DRlon)=0 THEN sights(current,i_DRlon):=DRlon_default; END;  
  IF sights(current,i_HS)=0 THEN sights(current,i_artificialFlag):=artificial_default; END;

  // If we already have a value for HS, set
  // default to not go through the observation
  // averaging process
  LOCAL avgReadings:=IFTE(sights(current,i_HS)=0,1,0);

  c:=INPUT({tempDate,
    sights(current,i_DRlat),
    sights(current,i_DRlon),         
    {sights(current,i_bodyIndex),bodiesList},
    {sights(current,i_artificialFlag),1},
    {avgReadings,1}},
    "LOP #"+STRING(current)+" Observation Parameters",
    {"UTC Date:","DR Lat:","DR Lon:","Body:","Artificial:","Average:"},
    {"UTC Date (YYYY.MMDD) (UTC Time: "+GET_UTC_TIME()+")",
     "DR Latitude ° (N=+ S=-)",
     "DR Longitude ° (N=+ S=-)",     
     "Observed body and limb",
     "Check if using an artificial horizon",
     "Perform observation averaging for HS"});
  IF NOT c THEN RETURN c; END;

  CASE
    IF sights(current,i_bodyIndex)=1 THEN sights(current,i_sightType):=1; sights(current,i_limbSign):=1; END;
    IF sights(current,i_bodyIndex)=2 THEN sights(current,i_sightType):=1; sights(current,i_limbSign):=2; END;
    IF sights(current,i_bodyIndex)=3 THEN sights(current,i_sightType):=2; sights(current,i_limbSign):=1; END;
    IF sights(current,i_bodyIndex)=4 THEN sights(current,i_sightType):=2; sights(current,i_limbSign):=2; END;
    IF (sights(current,i_bodyIndex)>=5) AND (sights(current,i_bodyIndex)<=8) THEN 
      sights(current,i_sightType):=4; 
      sights(current,i_subBodyIndex):=sights(current,i_bodyIndex)-4; 
    END;
    IF sights(current,i_bodyIndex)=9 THEN sights(current,i_sightType):=7; END;
    IF sights(current,i_bodyIndex)>9 THEN 
      sights(current,i_sightType):=3; 
      sights(current,i_subBodyIndex):=sights(current,i_bodyIndex)-9; 
    END;
    DEFAULT
      RETURN 0; 
  END;

  sights(current,i_DRlon):=NORMALIZE_LONGITUDE(sights(current,i_DRlon));
  sights(current,i_date):=tempDate;  
  DRlat_default:=sights(current,i_DRlat);
  DRlon_default:=sights(current,i_DRlon);
  artificial_default:=sights(current,i_artificialFlag);
  
  // If the pre-calculated option was selected,
  // skip the corrections and sextant altitude screens
  IF sights(current,i_sightType)=7 THEN RETURN 1; END;

  // OBSERVATION CORRECTIONS

  IF sights(current,i_indexCorrection_min)=0 THEN sights(current,i_indexCorrection_min):=IC_default; END;
  IF sights(current,i_eyeHeight_ft)=0 THEN sights(current,i_eyeHeight_ft):=HE_default; END;
  IF sights(current,i_temperature_C)=0 THEN sights(current,i_temperature_C):=temp_default; END;
  IF sights(current,i_pressure_mb)=0 THEN sights(current,i_pressure_mb):=press_default; END;
  
  c:=INPUT(
    {sights(current,i_indexCorrection_min),
     sights(current,i_eyeHeight_ft),
     sights(current,i_dipShort_nm),
     sights(current,i_temperature_C),
     sights(current,i_pressure_mb)},
    "LOP "+STRING(current)+": Corrections",
    {"IC:","HE:","Dip short:","Temp:","Pressure:"},
    {"Index Correction (') (+=Off -=On scale)",
     "Eye Height (ft.)",
     "Horizon distance (nm) (0 if true horizon)",
     "Temperature (°C)",
     "Pressure (mb)"});

  IF NOT c THEN RETURN c; END;
  
  IC_default:=sights(current,i_indexCorrection_min);
  HE_default:=sights(current,i_eyeHeight_ft);
  temp_default:=sights(current,i_temperature_C);
  press_default:=sights(current,i_pressure_mb);

  // OBSERVATION READINGS

  IF avgReadings=1 THEN
    c:=AVERAGE_READINGS(deltaUTCTime);
    IF SIZE(c)>0 THEN
      tempTime:=c(1);
      IF tempTime>24 THEN
        tempTime:=tempTime-24;
      ELSE
        IF tempTime<0 THEN
          tempTime:=tempTime+24;
        END;
      END;
      sights(current,i_time):=tempTime;
      sights(current,i_HS):=c(2);
    END;
  END;

  IF sights(current,i_time)=0 THEN sights(current,i_time):=tempTime; END;

  c:=INPUT({
    sights(current,i_date),
    sights(current,i_time),
    sights(current,i_HS)},
    "LOP #"+current+" "+bodiesList(sights(current,i_bodyIndex))+" Observation",
    {"Date:","Time:","HS:"},
    {"UTC Date (YYYY.MMDD)",
     "UTC Time (HH°MM'SS\")",
     "Sextant Height (DD°MM'SS\") - Uncorrected"}
  );
  
  IF c==0 OR sights(current,i_HS)==0 THEN RETURN 0; END;
  
  HA_CALC(current);  
  RETURN c;
END;


HA_CALC(index)
BEGIN

  // If a horizon distance is specified,
  // use dip short formula
  IF sights(index,i_dipShort_nm) > 0 THEN
    sights(index,i_dip_min):=(0.04158*sights(index,i_dipShort_nm))+(0.5658s*sights(index,i_eyeHeight_ft)/sights(index,i_dipShort_nm));
  ELSE
    sights(index,i_dip_min):=0.97*√(sights(index,i_eyeHeight_ft));
  END;
  IF sights(index,i_artificialFlag)==1 THEN
    sights(index,i_HA):=(sights(index,i_HS)+sights(index,i_indexCorrection_min)/60)/2;
  ELSE
    sights(index,i_HA):=sights(index,i_HS)+(sights(index,i_indexCorrection_min)-sights(index,i_dip_min))/60;
  END;

  RETURN sights(index,i_HA);
END;

LOCAL REFRACTION_MINUTES(P,T,a)
BEGIN
  LOCAL K:=273;  

  IF a<0 THEN RETURN 0; END;

  // Based on Bennett, G. G., J. Inst. Nav., vol. 30, p. 255
  // From https://navlist.net/img/104673.refraction.pdf
  RETURN 60*0.0167/TAN(a+7.31/(a+4.4))*0.28*P/(T+K);  
END;

LOCAL REFRACTION_MINUTES(index)
BEGIN
  LOCAL P:=sights(index,i_pressure_mb);
  LOCAL T:=sights(index,i_temperature_C);
  LOCAL a:=sights(index,i_HA);  
  
  RETURN REFRACTION_MINUTES(P,T,a);  
END;

HO_CALC_SUN(index,semiDiameter_min)
BEGIN

  sights(index,i_limb):=(sights(index,i_limbSign)-1)*2-1;  
  sights(index,i_semiDiameter_min):=semiDiameter_min;
  sights(index,i_refraction_min):=REFRACTION_MINUTES(index);
  sights(index,i_parallax_min):=0.15*COS(sights(index,i_HA));
  
  sights(index,i_altitudeCorr_min):=-sights(index,i_refraction_min)
    -sights(index,i_limb)*sights(index,i_semiDiameter_min)
    +sights(index,i_parallax_min);
  
  sights(index,i_HO):=sights(index,i_HA)
    +sights(index,i_altitudeCorr_min)/60;
  
  RETURN sights(index,i_HO);
END;

HC_CALC(index)
BEGIN

  LOCAL c:=HC_AZ(sights(index,i_GHA),
     sights(index,i_declination),
     sights(index,i_DRlat),
     sights(index,i_DRlon));
  
  sights(index,i_HC):=c(1);
  sights(index,i_Zn):=c(2);
  sights(index,i_LHA):=c(3);
  
  IF sights(index,i_HC)<0 THEN
    IF NOT MSGBOX("Hc is negative."+nl+
      "Unless you're looking through"+nl+
      "the earth, check your assumed"+nl+
      "position.",1) THEN RETURN 0; END;
  END;

  // Calculate distance offset from
  // assumed position in nautical miles
  sights(index,i_a_nm):=(sights(index,i_HO)-sights(index,i_HC))*60;

  RETURN sights(index,i_HC);
END;

LOCAL HC_AZ(gha,dec,lat,lon)
BEGIN
  // Calculate LHA from GHA 
  // based on longitude
  LOCAL lha:=(gha+lon) MOD 360;

  // Compute sign of declination and LHA 
  // to use Hc formula 
  LOCAL decSign:=SIGN(dec)*SIGN(lat);  
  LOCAL lhaSign:=−SIGN(lha-180);

  LOCAL hc:=ASIN(SIN(decSign*ABS(dec))
               *SIN(ABS(lat))
              +COS(decSign*ABS(dec))
               *COS(ABS(lat))
               *COS(lha)
          );

  LOCAL z:=ATAN2(SIN(lhaSign*lha),
              (COS(ABS(lat))
               *TAN(decSign*ABS(dec))
               -SIN(ABS(lat))
                *COS(lha))
          );

  // Adjust Z if negative
  IF z≤0 THEN
    z:=z+180;
  END;

  // Calculate azimuth to body based
  // on Z, latitude, and LHA
  IF lat≥0 THEN
    IF lha≥180 THEN
      z:=z;
    ELSE
      z:=360-z;
    END;
  ELSE
    IF lha≥180 THEN
      z:=180-z;
    ELSE
      z:=180+z;
    END;
  END; 

  RETURN {hc,z,lha};
END;

NORMALIZE_LONGITUDE(lon)
BEGIN
  LOCAL l:=lon MOD 360;
  IF l>180 THEN l:=l-360; END;
  RETURN l;
END;

// Returns the day number in the year for a
// given date. Used in calculations of sun semi-diameter.
DoY(date)
BEGIN
  LOCAL M:=IP(FP(date)*100);
  LOCAL D:=IP(FP(date*100)*100);

  IF M≤2 THEN
    RETURN IP(30.6*M+368.8)+D-399;
  ELSE
    RETURN IP(30.6*M+1.6)+D-34;
  END;
END;

LOCAL NM_DIST(lat1, lon1, lat2, lon2)
BEGIN
  LOCAL dLat, dLon, a, c, R;
  
  LOCAL oldHA:=HAngle;
  HAngle:=1;
  
  // Earth's radius in Nautical Miles
  R := 3440.065;
    
  // Differences
  dLat := lat2 - lat1;
  dLon := lon2 - lon1;
  
  // Haversine calculation
  a := SIN(dLat/2)^2 + COS(lat1) * COS(lat2) * SIN(dLon/2)^2;
  c := 2 * ATAN2(√(a), √(1-a)) * π/180;

  HAngle:=oldHA;
  
  RETURN R * c;
END;

// Returns the coordinates {longitude, latitude} of
// a point located d nautical miles from coordinates
// (latitude y1, longitude x1) on bearing of z degrees
// true.
// Inputs:  y0: initial latitude
//          x0: initial longitude
//      d : distance in nm
//      z : bearing in true degrees
//
// Returns: {longitude,latitude} : a list 
//          with the calculated coordinates

LOCAL atdz(y0,x0,d,z)
BEGIN
  LOCAL y1,x1,R;

  R:=3440.065;
   
  y1:=ASIN(SIN(y0)*COS((d/R)*(180/π))
      +COS(y0)*SIN((d/R)*(180/π))*COS(z));
  
  x1:=x0+ARG(i*SIN(z)*SIN((d/R)*(180/π))*COS(y0)+(COS((d/R)*(180/π))-SIN(y0)*SIN(y1)));
   
  RETURN {y1,x1};
END;

         +(time-12°0′0)/24;
END;

LOCAL HMS(H)

LOCAL ATAN2(y, x)
BEGIN
  RETURN ARG(x + i*y);
END;

LOCAL GET_UTC_DATETIME()
BEGIN
  LOCAL d:=Date;
  LOCAL t:=GET_UTC_TIME();
  
  IF t>24 THEN
    t:=t-24;
    d:=DATEADD(d,1);
  ELSE 
    IF t<0 THEN
      t:=t+24;
      d:=DATEADD(d,-1);
    END;
  END;
  
  RETURN {d,→HMS(t)};
END;

LOCAL GET_UTC_TIME()
BEGIN
  RETURN →HMS(Time - timeZone);
END;

LOCAL LOCAL_TIME(utcTime)
BEGIN
  LOCAL t:=utcTime+timeZone;
  IF t<0 THEN t:=t+24; END;
  IF t>=24 THEN t:=t-24; END;
  
  RETURN →HMS(t);
END;

GET_SOFTKEY(keylist)
BEGIN
  RETURN GET_SOFTKEY(keylist,{0,0,0,0,0,0});
END;

GET_SOFTKEY(keylist,flags)
BEGIN
  LOCAL k:=keylist,i;
  FOR i FROM 1 TO SIZE(flags) DO
    IF flags(i) THEN k(i):=CHAR(8226)+k(i); END;
  END;
  WHILE 1 DO
    DRAWMENU(k);
    LOCAL r:=WAIT(-1);
    // Check if we have a keyboard or
    // mouse event
    IF TYPE(r)=0 THEN
      // We have a keyboard event. Check if
      // either ESC or ON was pressed and
      // return 0 if so, and ignore any
      // other key
      IF r=4 OR r=46 THEN
        RETURN 0;
      END;
    ELSE 
      // We have a mouse event.
      // Check that we have a mouse 
      // click event 
      IF r(1)=3 THEN
        // Check that the click was
        // on the soft key bar at the
        // bottom
        IF r(3)>220 THEN
          // Check that the soft key
          // has a label attached to
          // it or do nothing
          LOCAL key:=IP(r(2)/53)+1;
          IF keylist(key)<>"" THEN
            RETURN key;
          END;
        END;
      END;
    END;
  END;
END;


// This function displays a menu of tiles
// with possible soft keys at the bottom and
// returns 0 if On or ESC was pressed, the index
// in the options list of the tile pressed
// or a the index of the softkeys list as a negative
// number if a soft key was pressed
//
// Arguments:
//     options       A list containing menu options
//     rows,columns  The layout of the tiles on the screen
//     softkeys      A list of softkeys. Empty 
//                   list ({}) for no soft keys
//
// Example usage:
// LOCAL c:=TILEMENU({
//   "1. Sights",
//   "2. Running Fix",
//   "3. Plot LOPs"},
//   5,2,
//   {"","","","","","Go Back"});
  
LOCAL TILEMENU(options,rows,columns,softkeys)
BEGIN
  LOCAL padding:=2;
  LOCAL height:=IFTE(SIZE(softkeys)<>0,220-padding,240);
  LOCAL itemHeight:=IP(height/rows);
  LOCAL width:=320;  
  LOCAL tileBorder:=0;
  LOCAL backgroundColor:=IFTE(Theme(1)==1,#FFFFFF,0);
  LOCAL tileColor:=IFTE(Theme(1)==1,#141D5C,#4A0909);
  LOCAL textColor:=#FFFFFF;
  
  LOCAL i:=1, j:=1, k:=1;
  RECT_P(0,0,width,320,backgroundColor,backgroundColor);
  
  WHILE k<=SIZE(options) DO
    LOCAL x0:=(i-1)*(width/columns)+padding;
  LOCAL x1:=i*(width/columns)-padding;
  LOCAL y0:=(j-1)*(itemHeight)+padding;
  LOCAL y1:=j*itemHeight-padding;
  IF options(k)<>"" THEN
    RECT_P(x0,y0,x1,y1,tileBorder,tileColor);
    TEXTOUT_P(options(k),x0+padding,y0+itemHeight/2-10,3,textColor,320,tileColor);
  END;
  k:=k+1;
  j:=j+1;
  IF j*itemHeight>height THEN
    j:=1;
    i:=i+1;
    IF i>columns THEN 
      k:=1e12;
    END;
  END;
  END;
  
  IF SIZE(softkeys)<>0 THEN
    DRAWMENU(softkeys);
  END;
  
  WHILE 1 DO
  LOCAL r:=WAIT(-1);
  // Check if we have a keyboard or
  // mouse event
  IF TYPE(r)=0 THEN
    // We have a keyboard event. Check if
    // either ESC or ON was pressed and
    // return 0 if so, and ignore any
    // other key
    IF r=4 OR r=46 THEN
    RETURN 0;
    END;
    
    // Find if a key corresponding to a menu
    // options was pressed    
    LOCAL key:=POS({47,42,43,44,37,38,39,32,33,34},r);    
    IF key<>0 THEN
      key:=key-1;    
    // Search for menu item starting with pressed number
    LOCAL i,r:=0;
    FOR i FROM 1 TO SIZE(options) DO      
      IF LEFT(options(i),1)==CHAR(48+key) THEN RETURN i; END;
    END;
    END;
  ELSE 
    // We have a mouse event.
    // Check that we have a mouse 
    // click event
    
    IF r(1)=3 THEN
      IF SIZE(softkeys)<>0 THEN
        // If we have softkeys, check
          // if a click was made on them
          IF r(3)>220 THEN
            // Check that the soft key
            // has a label attached to
            // it or do nothing
            LOCAL key:=IP(r(2)/53)+1;
            IF softkeys(key)<>"" THEN
              RETURN -1*key;
            END;
          END;
    END;
      // Calculate which menu item    
    // was clicked
    LOCAL col:=IP(r(2)/(width/columns));
    LOCAL row:=IP(r(3)/(height/rows));
    LOCAL item:=col*rows+row+1;
    IF item<=SIZE(options) THEN
      RETURN item;
    END;
    END;
  END;
  END;
END;
